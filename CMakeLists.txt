CMAKE_MINIMUM_REQUIRED(VERSION 3.10)
PROJECT(subconverter LANGUAGES CXX)
SET(BUILD_TARGET_NAME ${PROJECT_NAME})
SET(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CMAKE_SOURCE_DIR}/cmake/")
INCLUDE_DIRECTORIES("${CMAKE_SOURCE_DIR}/include/")

SET(SUBCONVERTER_PROJECT_COMMIT "" CACHE STRING "Commit used for this project build")
SET(SUBCONVERTER_BUILD_VERSION "" CACHE STRING "Override the complete build version")
SET(SUBCONVERTER_MIHOMO_FETCHER_BINARY "" CACHE FILEPATH "Locked Mihomo helper installed beside subconverter")
SET(SUBCONVERTER_MIHOMO_FETCHER_MANIFEST "" CACHE FILEPATH "Manifest for the locked Mihomo helper")
SET(SUBCONVERTER_MIHOMO_FETCHER_PLATFORM "" CACHE STRING "Platform identity of the locked Mihomo helper")

FILE(READ "${CMAKE_SOURCE_DIR}/.github/project-metadata.json" PROJECT_METADATA_JSON)
FILE(READ "${CMAKE_SOURCE_DIR}/.github/source-lock.json" SOURCE_LOCK_JSON)
STRING(REGEX MATCH "\"upstream_commit\"[ \t\r\n]*:[ \t\r\n]*\"([0-9a-fA-F]+)\"" _ "${PROJECT_METADATA_JSON}")
SET(SUBCONVERTER_UPSTREAM_COMMIT "${CMAKE_MATCH_1}")
STRING(REGEX MATCH "\"upstream_version\"[ \t\r\n]*:[ \t\r\n]*\"([^\"]+)\"" _ "${PROJECT_METADATA_JSON}")
SET(SUBCONVERTER_METADATA_UPSTREAM_VERSION "${CMAKE_MATCH_1}")
STRING(REGEX MATCH "\"edition\"[ \t\r\n]*:[ \t\r\n]*\"([A-Za-z0-9._-]+)\"" _ "${PROJECT_METADATA_JSON}")
SET(SUBCONVERTER_EDITION "${CMAKE_MATCH_1}")

STRING(REGEX MATCH "\"tag\"[ \t\r\n]*:[ \t\r\n]*\"(v[0-9]+\\.[0-9]+\\.[0-9]+)\"" _ "${SOURCE_LOCK_JSON}")
SET(SUBCONVERTER_MIHOMO_VERSION "${CMAKE_MATCH_1}")
STRING(REGEX MATCH "\"tag_identity\"[^{]*\{[ \t\r\n]*\"annotated_tags\"[^}]*\"commit\"[ \t\r\n]*:[ \t\r\n]*\"([0-9a-fA-F]+)\"" _ "${SOURCE_LOCK_JSON}")
SET(SUBCONVERTER_MIHOMO_COMMIT "${CMAKE_MATCH_1}")
STRING(REGEX MATCH "\"helper_overlay_sha256\"[ \t\r\n]*:[ \t\r\n]*\"(sha256:[0-9a-fA-F]+)\"" _ "${SOURCE_LOCK_JSON}")
SET(SUBCONVERTER_MIHOMO_OVERLAY_SHA256 "${CMAKE_MATCH_1}")
STRING(REGEX MATCH "\"pair_id\"[ \t\r\n]*:[ \t\r\n]*\"(sha256:[0-9a-fA-F]+)\"" _ "${SOURCE_LOCK_JSON}")
SET(SUBCONVERTER_MIHOMO_PAIR_ID "${CMAKE_MATCH_1}")
STRING(REGEX MATCH "\"helper_protocol\"[ \t\r\n]*:[ \t\r\n]*([0-9]+)" _ "${SOURCE_LOCK_JSON}")
SET(SUBCONVERTER_MIHOMO_PROTOCOL "${CMAKE_MATCH_1}")
STRING(REGEX MATCH "\"parity_contract\"[ \t\r\n]*:[ \t\r\n]*\"([^\"]+)\"" _ "${SOURCE_LOCK_JSON}")
SET(SUBCONVERTER_MIHOMO_PARITY_CONTRACT "${CMAKE_MATCH_1}")

IF(NOT SUBCONVERTER_UPSTREAM_COMMIT OR NOT SUBCONVERTER_METADATA_UPSTREAM_VERSION OR NOT SUBCONVERTER_EDITION)
    MESSAGE(FATAL_ERROR "Invalid .github/project-metadata.json")
ENDIF()
IF(NOT SUBCONVERTER_MIHOMO_VERSION OR NOT SUBCONVERTER_MIHOMO_COMMIT OR
   NOT SUBCONVERTER_MIHOMO_OVERLAY_SHA256 OR NOT SUBCONVERTER_MIHOMO_PAIR_ID OR
   NOT SUBCONVERTER_MIHOMO_PROTOCOL OR NOT SUBCONVERTER_MIHOMO_PARITY_CONTRACT)
    MESSAGE(FATAL_ERROR "Invalid .github/source-lock.json")
ENDIF()

SET(SUBCONVERTER_MIHOMO_HELPER_SHA256 "")
SET(SUBCONVERTER_MIHOMO_HELPER_PLATFORM "")
SET(SUBCONVERTER_MIHOMO_HELPER_NAME "")
IF(SUBCONVERTER_MIHOMO_FETCHER_BINARY OR SUBCONVERTER_MIHOMO_FETCHER_MANIFEST OR
   SUBCONVERTER_MIHOMO_FETCHER_PLATFORM)
    IF(NOT SUBCONVERTER_MIHOMO_FETCHER_BINARY OR NOT SUBCONVERTER_MIHOMO_FETCHER_MANIFEST OR
       NOT SUBCONVERTER_MIHOMO_FETCHER_PLATFORM)
        MESSAGE(FATAL_ERROR "Mihomo helper binary, manifest, and platform must be supplied together")
    ENDIF()
    IF(NOT EXISTS "${SUBCONVERTER_MIHOMO_FETCHER_BINARY}" OR
       NOT EXISTS "${SUBCONVERTER_MIHOMO_FETCHER_MANIFEST}")
        MESSAGE(FATAL_ERROR "Configured Mihomo helper or manifest does not exist")
    ENDIF()

    FILE(READ "${SUBCONVERTER_MIHOMO_FETCHER_MANIFEST}" MIHOMO_HELPER_MANIFEST_JSON)
    FILE(SHA256 "${SUBCONVERTER_MIHOMO_FETCHER_BINARY}" MIHOMO_HELPER_COMPUTED_SHA256)
    STRING(REGEX MATCH "\"schema_version\"[ \t\r\n]*:[ \t\r\n]*([0-9]+)" _ "${MIHOMO_HELPER_MANIFEST_JSON}")
    SET(MIHOMO_HELPER_SCHEMA "${CMAKE_MATCH_1}")
    STRING(REGEX MATCH "\"pair_id\"[ \t\r\n]*:[ \t\r\n]*\"(sha256:[0-9a-fA-F]+)\"" _ "${MIHOMO_HELPER_MANIFEST_JSON}")
    SET(MIHOMO_HELPER_PAIR_ID "${CMAKE_MATCH_1}")
    STRING(REGEX MATCH "\"platform\"[ \t\r\n]*:[ \t\r\n]*\"([^\"]+)\"" _ "${MIHOMO_HELPER_MANIFEST_JSON}")
    SET(MIHOMO_HELPER_PLATFORM "${CMAKE_MATCH_1}")

    STRING(REGEX MATCH "\"helper\"[ \t\r\n]*:[ \t\r\n]*\\{[^}]*\\}" MIHOMO_HELPER_BLOCK "${MIHOMO_HELPER_MANIFEST_JSON}")
    STRING(REGEX MATCH "\"name\"[ \t\r\n]*:[ \t\r\n]*\"([^\"]+)\"" _ "${MIHOMO_HELPER_BLOCK}")
    SET(MIHOMO_HELPER_NAME "${CMAKE_MATCH_1}")
    STRING(REGEX MATCH "\"sha256\"[ \t\r\n]*:[ \t\r\n]*\"(sha256:[0-9a-fA-F]+)\"" _ "${MIHOMO_HELPER_BLOCK}")
    SET(MIHOMO_HELPER_DECLARED_SHA256 "${CMAKE_MATCH_1}")

    STRING(REGEX MATCH "\"mihomo\"[ \t\r\n]*:[ \t\r\n]*\\{[^}]*\\}" MIHOMO_MANIFEST_IDENTITY_BLOCK "${MIHOMO_HELPER_MANIFEST_JSON}")
    STRING(REGEX MATCH "\"commit\"[ \t\r\n]*:[ \t\r\n]*\"([0-9a-fA-F]+)\"" _ "${MIHOMO_MANIFEST_IDENTITY_BLOCK}")
    SET(MIHOMO_MANIFEST_COMMIT "${CMAKE_MATCH_1}")
    STRING(REGEX MATCH "\"tag\"[ \t\r\n]*:[ \t\r\n]*\"([^\"]+)\"" _ "${MIHOMO_MANIFEST_IDENTITY_BLOCK}")
    SET(MIHOMO_MANIFEST_TAG "${CMAKE_MATCH_1}")

    STRING(REGEX MATCH "\"project\"[ \t\r\n]*:[ \t\r\n]*\\{[^}]*\\}" MIHOMO_MANIFEST_PROJECT_BLOCK "${MIHOMO_HELPER_MANIFEST_JSON}")
    STRING(REGEX MATCH "\"helper_overlay_sha256\"[ \t\r\n]*:[ \t\r\n]*\"(sha256:[0-9a-fA-F]+)\"" _ "${MIHOMO_MANIFEST_PROJECT_BLOCK}")
    SET(MIHOMO_MANIFEST_OVERLAY "${CMAKE_MATCH_1}")
    STRING(REGEX MATCH "\"helper_protocol\"[ \t\r\n]*:[ \t\r\n]*([0-9]+)" _ "${MIHOMO_MANIFEST_PROJECT_BLOCK}")
    SET(MIHOMO_MANIFEST_PROTOCOL "${CMAKE_MATCH_1}")
    STRING(REGEX MATCH "\"parity_contract\"[ \t\r\n]*:[ \t\r\n]*\"([^\"]+)\"" _ "${MIHOMO_MANIFEST_PROJECT_BLOCK}")
    SET(MIHOMO_MANIFEST_PARITY_CONTRACT "${CMAKE_MATCH_1}")

    STRING(TOLOWER "${MIHOMO_HELPER_COMPUTED_SHA256}" MIHOMO_HELPER_COMPUTED_SHA256)
    STRING(TOLOWER "${MIHOMO_HELPER_DECLARED_SHA256}" MIHOMO_HELPER_DECLARED_SHA256)
    IF(NOT MIHOMO_HELPER_SCHEMA STREQUAL "1" OR
       NOT MIHOMO_HELPER_PAIR_ID STREQUAL SUBCONVERTER_MIHOMO_PAIR_ID OR
       NOT MIHOMO_HELPER_PLATFORM STREQUAL SUBCONVERTER_MIHOMO_FETCHER_PLATFORM OR
       NOT MIHOMO_MANIFEST_TAG STREQUAL SUBCONVERTER_MIHOMO_VERSION OR
       NOT MIHOMO_MANIFEST_COMMIT STREQUAL SUBCONVERTER_MIHOMO_COMMIT OR
       NOT MIHOMO_MANIFEST_OVERLAY STREQUAL SUBCONVERTER_MIHOMO_OVERLAY_SHA256 OR
       NOT MIHOMO_MANIFEST_PROTOCOL STREQUAL SUBCONVERTER_MIHOMO_PROTOCOL OR
       NOT MIHOMO_MANIFEST_PARITY_CONTRACT STREQUAL SUBCONVERTER_MIHOMO_PARITY_CONTRACT OR
       NOT MIHOMO_HELPER_DECLARED_SHA256 STREQUAL "sha256:${MIHOMO_HELPER_COMPUTED_SHA256}")
        MESSAGE(FATAL_ERROR "Mihomo helper manifest does not match the binary and source lock")
    ENDIF()
    IF(SUBCONVERTER_MIHOMO_FETCHER_PLATFORM MATCHES "^windows-")
        SET(MIHOMO_EXPECTED_HELPER_NAME "subconverter-mihomo-fetcher.exe")
    ELSE()
        SET(MIHOMO_EXPECTED_HELPER_NAME "subconverter-mihomo-fetcher")
    ENDIF()
    IF(NOT MIHOMO_HELPER_NAME STREQUAL MIHOMO_EXPECTED_HELPER_NAME)
        MESSAGE(FATAL_ERROR "Mihomo helper manifest name does not match its platform")
    ENDIF()

    SET(SUBCONVERTER_MIHOMO_HELPER_SHA256 "sha256:${MIHOMO_HELPER_COMPUTED_SHA256}")
    SET(SUBCONVERTER_MIHOMO_HELPER_PLATFORM "${MIHOMO_HELPER_PLATFORM}")
    SET(SUBCONVERTER_MIHOMO_HELPER_NAME "${MIHOMO_HELPER_NAME}")
ENDIF()

FILE(STRINGS "${CMAKE_SOURCE_DIR}/src/version.h" UPSTREAM_VERSION_LINE REGEX "^#define VERSION ")
STRING(REGEX REPLACE ".*\"([^\"]+)\".*" "\\1" SUBCONVERTER_UPSTREAM_VERSION "${UPSTREAM_VERSION_LINE}")
IF(NOT SUBCONVERTER_UPSTREAM_VERSION STREQUAL SUBCONVERTER_METADATA_UPSTREAM_VERSION)
    MESSAGE(FATAL_ERROR "Upstream version metadata does not match src/version.h")
ENDIF()

STRING(SUBSTRING "${SUBCONVERTER_UPSTREAM_COMMIT}" 0 8 SUBCONVERTER_UPSTREAM_COMMIT_SHORT)

IF(NOT SUBCONVERTER_PROJECT_COMMIT)
    EXECUTE_PROCESS(
        COMMAND git status --porcelain --untracked-files=normal
        WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
        RESULT_VARIABLE GIT_STATUS_RESULT
        OUTPUT_VARIABLE GIT_STATUS_OUTPUT
        OUTPUT_STRIP_TRAILING_WHITESPACE
        ERROR_QUIET)
    IF(GIT_STATUS_RESULT EQUAL 0 AND NOT GIT_STATUS_OUTPUT)
        EXECUTE_PROCESS(
            COMMAND git rev-parse --short=8 HEAD
            WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
            OUTPUT_VARIABLE SUBCONVERTER_PROJECT_COMMIT
            OUTPUT_STRIP_TRAILING_WHITESPACE
            ERROR_QUIET)
    ENDIF()
ENDIF()

IF(NOT SUBCONVERTER_PROJECT_COMMIT)
    SET(SUBCONVERTER_PROJECT_COMMIT "local")
ENDIF()

IF(SUBCONVERTER_PROJECT_COMMIT MATCHES "^[0-9a-fA-F]+$")
    STRING(LENGTH "${SUBCONVERTER_PROJECT_COMMIT}" PROJECT_COMMIT_LENGTH)
    IF(PROJECT_COMMIT_LENGTH GREATER 8)
        STRING(SUBSTRING "${SUBCONVERTER_PROJECT_COMMIT}" 0 8 SUBCONVERTER_PROJECT_COMMIT_SHORT)
    ELSE()
        SET(SUBCONVERTER_PROJECT_COMMIT_SHORT "${SUBCONVERTER_PROJECT_COMMIT}")
    ENDIF()
ELSE()
    SET(SUBCONVERTER_PROJECT_COMMIT_SHORT "${SUBCONVERTER_PROJECT_COMMIT}")
ENDIF()

        "${SUBCONVERTER_UPSTREAM_VERSION}-${SUBCONVERTER_UPSTREAM_COMMIT_SHORT}-${SUBCONVERTER_EDITION}.${SUBCONVERTER_PROJECT_COMMIT_SHORT}")
ENDIF()
